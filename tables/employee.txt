
CREATE TABLE employee (
    emp_id NUMBER GENERATED ALWAYS AS IDENTITY START WITH 1 INCREMENT BY 1 PRIMARY KEY,
    emp_name VARCHAR2(50) NOT NULL,
    dept_id number not null,
    salary number,
    manager_id number not null
);
